class Solution {
  public:
    int  check(vector<int>&nums,vector<int>&dp,int n,int i){
        if(i>=(n-1)) return 0;
        if(dp[i]!=-1) return dp[i];
        int left=check(nums,dp,n,i+1);
        int right=check(nums,dp,n,i+2);
        dp[i]=min((abs(nums[i]-nums[i+1])+left),(abs(nums[i]-nums[i+2])+right));
        return dp[i];
    }
    int minCost(vector<int>& nums) {
        // Code here
        int n=nums.size();
        vector<int>dp(n,-1);
        nums.push_back(INT_MAX);
        dp[n-1]=0;
        int ans=check(nums,dp,n,0);
        return ans;
    }
};
